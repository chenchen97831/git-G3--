name: Presentation_Generation_Graph
description: 兒童導向與嚴格 RAG 限制的簡報生成產線
max_global_loops: 3

nodes:
  - id: card_generator
    type: llm_agent
    input: [sources.md]
    output: [cards/]
  
  - id: outline_generator
    type: llm_agent
    input: [cards/]
    output: [outline.md]

  - id: drafter
    type: llm_agent
    input: [outline.md, cards/]
    output: [drafts/]
    max_retries: 2

  - id: validate_drafts
    type: script_runner
    scripts:
      - "python scripts/check_word_count.py {target_file}"
      - "python scripts/check_sources.py {target_file}"

  - id: renderer
    type: code_agent
    input: [drafts/]
    output: [deck.html, report.md]
    max_retries: 3

  - id: validate_render
    type: script_runner
    scripts:
      - "python scripts/check_font_size.py deck.html"
      - "python scripts/check_sources.py deck.html"
      - "python scripts/check_external_links.py deck.html"

edges:
  - from: start
    to: card_generator
  - from: card_generator
    to: outline_generator
  - from: outline_generator
    to: drafter
  - from: drafter
    to: validate_drafts
  - from: validate_drafts
    condition: failed
    target: drafter
    action: target_retry
  - from: validate_drafts
    condition: passed
    target: renderer
  - from: renderer
    to: validate_render
  - from: validate_render
    condition: failed
    target: renderer
    action: visual_retry
  - from: validate_render
    condition: passed
    target: end

interrupts:
  - condition: "data_insufficient"
    action: halt_and_notify_user
    message: "資料卡不足以支撐大綱，請補充 sources，嚴禁自行發明內容。"
